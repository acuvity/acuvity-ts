# Mcpmessage

Represents MCP message details.

## Example Usage

```typescript
import { Mcpmessage } from "@acuvity/acuvity/models/components";

let value: Mcpmessage = {
  direction: "Client2Server",
  method: "tools/call",
  requestID: "2",
  sessionID: "1f02aa20-22d8-6e87-8432-be15d4f7b5b2",
  type: "Request",
};
```
